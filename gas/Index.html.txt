<script>/* ===== Crumb analysis & statistics engine (pure JS, no DOM) ===== */
const Engine = (() => {
  // ---------- colour ----------
  function toLab(rgba, n) {
    const L = new Float32Array(n), A = new Float32Array(n), B = new Float32Array(n);
    const lin = new Float32Array(256);
    for (let i = 0; i < 256; i++) { const c = i / 255; lin[i] = c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4); }
    const f = t => t > 0.008856 ? Math.cbrt(t) : 7.787 * t + 16 / 116;
    for (let i = 0; i < n; i++) {
      const r = lin[rgba[4 * i]], g = lin[rgba[4 * i + 1]], b = lin[rgba[4 * i + 2]];
      const x = (0.4124 * r + 0.3576 * g + 0.1805 * b) / 0.95047, y = 0.2126 * r + 0.7152 * g + 0.0722 * b, z = (0.0193 * r + 0.1192 * g + 0.9505 * b) / 1.08883;
      const fx = f(x), fy = f(y), fz = f(z);
      L[i] = 116 * fy - 16; A[i] = 500 * (fx - fy); B[i] = 200 * (fy - fz);
    }
    return { L, A, B };
  }
  // ---------- filters ----------
  function boxBlur(src, w, h, r) {
    const tmp = new Float32Array(w * h), out = new Float32Array(w * h), d = 2 * r + 1;
    for (let y = 0; y < h; y++) {
      let s = 0; const o = y * w;
      for (let x = -r; x <= r; x++) s += src[o + Math.min(w - 1, Math.max(0, x))];
      for (let x = 0; x < w; x++) { tmp[o + x] = s / d; s += src[o + Math.min(w - 1, x + r + 1)] - src[o + Math.max(0, x - r)]; }
    }
    for (let x = 0; x < w; x++) {
      let s = 0;
      for (let y = -r; y <= r; y++) s += tmp[Math.min(h - 1, Math.max(0, y)) * w + x];
      for (let y = 0; y < h; y++) { out[y * w + x] = s / d; s += tmp[Math.min(h - 1, y + r + 1) * w + x] - tmp[Math.max(0, y - r) * w + x]; }
    }
    return out;
  }
  const gauss = (src, w, h, s) => { const r = Math.max(1, Math.round(s * 0.9)); return boxBlur(boxBlur(boxBlur(src, w, h, r), w, h, r), w, h, r); };
  function minmax(src, w, h, r, isMax) { // separable square min/max filter
    const tmp = new Float32Array(w * h), out = new Float32Array(w * h), op = isMax ? Math.max : Math.min;
    for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) { let v = src[y * w + x]; for (let k = Math.max(0, x - r); k <= Math.min(w - 1, x + r); k++) v = op(v, src[y * w + k]); tmp[y * w + x] = v; }
    for (let x = 0; x < w; x++) for (let y = 0; y < h; y++) { let v = tmp[y * w + x]; for (let k = Math.max(0, y - r); k <= Math.min(h - 1, y + r); k++) v = op(v, tmp[k * w + x]); out[y * w + x] = v; }
    return out;
  }
  function otsu(vals) {
    let mn = Infinity, mx = -Infinity; for (const v of vals) { if (v < mn) mn = v; if (v > mx) mx = v; }
    const bins = 256, hist = new Float64Array(bins), sc = (bins - 1) / (mx - mn || 1);
    for (const v of vals) hist[Math.round((v - mn) * sc)]++;
    let tot = vals.length, sum = 0; for (let i = 0; i < bins; i++) sum += i * hist[i];
    let wB = 0, sB = 0, best = 0, th = 0;
    for (let i = 0; i < bins; i++) { wB += hist[i]; if (!wB) continue; const wF = tot - wB; if (!wF) break; sB += i * hist[i]; const mB = sB / wB, mF = (sum - sB) / wF, v = wB * wF * (mB - mF) ** 2; if (v > best) { best = v; th = i; } }
    return mn + th / sc;
  }
  function components(mask, w, h) { // 4-connected labelling; returns {lab, sizes}
    const lab = new Int32Array(w * h), sizes = [0], st = new Int32Array(w * h);
    let n = 0;
    for (let i = 0; i < w * h; i++) {
      if (!mask[i] || lab[i]) continue;
      n++; let sp = 0, cnt = 0; st[sp++] = i; lab[i] = n;
      while (sp) { const p = st[--sp]; cnt++; const x = p % w, y = (p / w) | 0;
        if (x > 0 && mask[p - 1] && !lab[p - 1]) { lab[p - 1] = n; st[sp++] = p - 1; }
        if (x < w - 1 && mask[p + 1] && !lab[p + 1]) { lab[p + 1] = n; st[sp++] = p + 1; }
        if (y > 0 && mask[p - w] && !lab[p - w]) { lab[p - w] = n; st[sp++] = p - w; }
        if (y < h - 1 && mask[p + w] && !lab[p + w]) { lab[p + w] = n; st[sp++] = p + w; } }
      sizes.push(cnt);
    }
    return { lab, sizes };
  }
  function fillHoles(mask, w, h) {
    const inv = new Uint8Array(w * h); for (let i = 0; i < w * h; i++) inv[i] = mask[i] ? 0 : 1;
    const { lab } = components(inv, w, h), edge = new Set();
    for (let x = 0; x < w; x++) { edge.add(lab[x]); edge.add(lab[(h - 1) * w + x]); }
    for (let y = 0; y < h; y++) { edge.add(lab[y * w]); edge.add(lab[y * w + w - 1]); }
    const out = new Uint8Array(w * h); for (let i = 0; i < w * h; i++) out[i] = mask[i] || !edge.has(lab[i]) ? 1 : 0;
    return out;
  }
  const erode = (m, w, h, r) => { const f = new Float32Array(m); const o = minmax(f, w, h, r, false); const u = new Uint8Array(w * h); for (let i = 0; i < w * h; i++) u[i] = o[i] > 0.5; return u; };
  const dilate = (m, w, h, r) => { const f = new Float32Array(m); const o = minmax(f, w, h, r, true); const u = new Uint8Array(w * h); for (let i = 0; i < w * h; i++) u[i] = o[i] > 0.5; return u; };

  // ---------- segmentation: single loaf on a light board ----------
  function segment(lab, w, h) {
    const bs = gauss(lab.B, w, h, 2), vals = [];
    for (let i = 0; i < w * h; i += 3) vals.push(bs[i]);
    const th = otsu(vals);
    let m = new Uint8Array(w * h);
    for (let i = 0; i < w * h; i++) m[i] = bs[i] > th && lab.L[i] > 35 ? 1 : 0;
    m = dilate(erode(m, w, h, 2), w, h, 2);
    const { lab: cl, sizes } = components(m, w, h);
    let best = 0; for (let i = 1; i < sizes.length; i++) if (sizes[i] > (sizes[best] || 0)) best = i;
    const out = new Uint8Array(w * h); for (let i = 0; i < w * h; i++) out[i] = cl[i] === best ? 1 : 0;
    return fillHoles(out, w, h);
  }

  // ---------- full analysis of one loaf image ----------
  // img = {data: RGBA Uint8ClampedArray, width, height}; resample: function(img, box, maxDim) -> img (provided by caller)
  function analyze(img, resample) {
    // pass 1: low-res segmentation to locate the loaf
    const small = resample(img, null, 360);
    const sl = toLab(small.data, small.width * small.height);
    const sm = segment(sl, small.width, small.height);
    let x0 = 1e9, y0 = 1e9, x1 = -1, y1 = -1, area = 0;
    for (let y = 0; y < small.height; y++) for (let x = 0; x < small.width; x++) if (sm[y * small.width + x]) { area++; if (x < x0) x0 = x; if (x > x1) x1 = x; if (y < y0) y0 = y; if (y > y1) y1 = y; }
    const fracArea = area / (small.width * small.height);
    const q = { loafArea: fracArea };
    if (area < 50) return { ok: false, reason: 'لم يتم العثور على رغيف في الصورة', quality: q };
    const s = img.width / small.width, pad = 6;
    const box = { x: Math.max(0, (x0 - pad) * s), y: Math.max(0, (y0 - pad) * s), w: Math.min(img.width, (x1 + pad) * s) - Math.max(0, (x0 - pad) * s), h: Math.min(img.height, (y1 + pad) * s) - Math.max(0, (y0 - pad) * s) };
    // pass 2: normalised crop (loaf ~ 600 px) => scale-invariant features
    const c = resample(img, box, 600), w = c.width, h = c.height, n = w * h;
    const lab = toLab(c.data, n);
    let mask = segment(lab, w, h);
    const D = Math.sqrt(mask.reduce((a, b) => a + b, 0) * 4 / Math.PI);       // equivalent diameter (px)
    const rim = Math.max(3, Math.round(D * 0.035));
    const core = erode(mask, w, h, rim);
    let N = 0; for (let i = 0; i < n; i++) N += core[i];
    if (N < 2000) return { ok: false, reason: 'الرغيف صغير جداً في الصورة — قرّب الجوال', quality: q };
    // illumination-normalised lightness
    const bg = gauss(lab.L, w, h, Math.max(8, D / 12));
    const Ln = new Float32Array(n); for (let i = 0; i < n; i++) Ln[i] = lab.L[i] / (bg[i] + 1e-3);
    // quality metrics
    let sL = 0, clip = 0, lapV = 0, lapM = 0;
    for (let y = 1; y < h - 1; y++) for (let x = 1; x < w - 1; x++) { const i = y * w + x; if (!core[i]) continue; sL += lab.L[i]; if (lab.L[i] > 97) clip++; const lp = 4 * lab.L[i] - lab.L[i - 1] - lab.L[i + 1] - lab.L[i - w] - lab.L[i + w]; lapM += lp; lapV += lp * lp; }
    q.meanL = sL / N; q.clipped = clip / N; lapM /= N; q.sharpness = lapV / N - lapM * lapM;
    // pores: morphological black-hat at the pore scale
    const k = Math.max(2, Math.round(D / 85));
    const Ls = new Float32Array(n); for (let i = 0; i < n; i++) Ls[i] = Ln[i] * 120;
    const close = minmax(minmax(Ls, w, h, k, true), w, h, k, false);
    const pore = new Uint8Array(n); let pc = 0;
    for (let i = 0; i < n; i++) { const bh = close[i] - Ls[i]; pore[i] = core[i] && bh > 14 ? 1 : 0; pc += pore[i]; }
    const poreFrac = pc / N;
    const { sizes: ps } = components(pore, w, h); const pSizes = ps.slice(1).filter(v => v >= 2);
    const poreDensity = pSizes.length / N * 10000;                              // pores per 10k px
    const pm = pSizes.length ? pSizes.reduce((a, b) => a + b, 0) / pSizes.length : 0;
    const psd = pSizes.length ? Math.sqrt(pSizes.reduce((a, b) => a + (b - pm) ** 2, 0) / pSizes.length) : 0;
    const poreSizeCV = pm ? psd / pm : 0;
    // local texture (std) -> smooth fraction
    const r2 = Math.max(3, Math.round(D / 60));
    const mu = boxBlur(Ln, w, h, r2), mu2 = boxBlur(Ln.map(v => v * v), w, h, r2);
    let smooth = 0, texSum = 0; const tex = new Float32Array(n);
    for (let i = 0; i < n; i++) { if (!core[i]) continue; const sd = Math.sqrt(Math.max(0, mu2[i] - mu[i] * mu[i])); tex[i] = sd; texSum += sd; if (sd < 0.018) smooth++; }
    const smoothFrac = smooth / N, texMean = texSum / N;
    // wet / grey patches: darker AND greyer than loaf median, large blobs only
    const Lb = gauss(lab.L, w, h, Math.max(4, D / 40)), Bb = gauss(lab.B, w, h, Math.max(4, D / 40));
    const lv = [], bv = []; for (let i = 0; i < n; i += 2) if (core[i]) { lv.push(Lb[i]); bv.push(Bb[i]); }
    lv.sort((a, b) => a - b); bv.sort((a, b) => a - b);
    const Lmed = lv[lv.length >> 1], Bmed = bv[bv.length >> 1];
    const wetC = new Uint8Array(n); for (let i = 0; i < n; i++) wetC[i] = core[i] && Lb[i] < Lmed - 12 && Bb[i] < Bmed - 4 ? 1 : 0;
    const { lab: wl, sizes: ws } = components(wetC, w, h);
    const wet = new Uint8Array(n); let wc = 0;
    for (let i = 0; i < n; i++) if (wl[i] && ws[wl[i]] > 0.01 * N) { wet[i] = 1; wc++; }
    const wetFrac = wc / N;
    // spatial uniformity: CV of pore fraction over a 5x5 grid
    let bx0 = w, by0 = h, bx1 = 0, by1 = 0; for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) if (core[y * w + x]) { if (x < bx0) bx0 = x; if (x > bx1) bx1 = x; if (y < by0) by0 = y; if (y > by1) by1 = y; }
    const G = 5, cells = [];
    for (let gy = 0; gy < G; gy++) for (let gx = 0; gx < G; gx++) {
      let c1 = 0, c2 = 0; const ya = by0 + ((by1 - by0) * gy / G) | 0, yb = by0 + ((by1 - by0) * (gy + 1) / G) | 0, xa = bx0 + ((bx1 - bx0) * gx / G) | 0, xb = bx0 + ((bx1 - bx0) * (gx + 1) / G) | 0;
      for (let y = ya; y < yb; y++) for (let x = xa; x < xb; x++) { const i = y * w + x; if (core[i]) { c1++; c2 += pore[i]; } }
      if (c1 > 0.7 * (yb - ya) * (xb - xa)) cells.push(c2 / c1);
    }
    const cm = cells.reduce((a, b) => a + b, 0) / (cells.length || 1);
    const gridCV = cm < 0.005 ? 1 : cells.length > 3 ? Math.sqrt(cells.reduce((a, b) => a + (b - cm) ** 2, 0) / cells.length) / (cm + 1e-6) : 0;
    // lacunarity of pore mask (gliding box, integral image)
    const ii = new Float64Array((w + 1) * (h + 1));
    for (let y = 0; y < h; y++) { let row = 0; for (let x = 0; x < w; x++) { row += pore[y * w + x]; ii[(y + 1) * (w + 1) + x + 1] = ii[y * (w + 1) + x + 1] + row; } }
    const bxs = Math.max(8, Math.round(D / 12)); let m1 = 0, m2 = 0, nb = 0;
    for (let y = 0; y + bxs < h; y += 2) for (let x = 0; x + bxs < w; x += 2) {
      if (!core[(y + (bxs >> 1)) * w + x + (bxs >> 1)] || !core[y * w + x] || !core[(y + bxs) * w + x + bxs]) continue;
      const M = ii[(y + bxs) * (w + 1) + x + bxs] - ii[y * (w + 1) + x + bxs] - ii[(y + bxs) * (w + 1) + x] + ii[y * (w + 1) + x];
      m1 += M; m2 += M * M; nb++;
    }
    const lacunarity = nb && m1 ? Math.min(50, (m2 / nb) / ((m1 / nb) ** 2)) : 50;
    // GLCM (16 grey levels, distance ~ pore scale, 4 directions)
    const Q = 16; let qmn = Infinity, qmx = -Infinity;
    for (let i = 0; i < n; i++) if (core[i]) { if (Ln[i] < qmn) qmn = Ln[i]; if (Ln[i] > qmx) qmx = Ln[i]; }
    const qv = new Uint8Array(n); for (let i = 0; i < n; i++) qv[i] = Math.min(Q - 1, Math.max(0, Math.floor((Ln[i] - qmn) / (qmx - qmn + 1e-6) * Q)));
    const P = new Float64Array(Q * Q); let pt = 0; const dd = Math.max(1, k);
    for (const [dx, dy] of [[dd, 0], [0, dd], [dd, dd], [dd, -dd]])
      for (let y = Math.max(0, -dy); y < h - Math.max(0, dy); y++) for (let x = 0; x < w - dx; x++) { const i = y * w + x, j = (y + dy) * w + x + dx; if (core[i] && core[j]) { P[qv[i] * Q + qv[j]]++; P[qv[j] * Q + qv[i]]++; pt += 2; } }
    let con = 0, hom = 0, ene = 0, ent = 0;
    for (let a = 0; a < Q; a++) for (let b = 0; b < Q; b++) { const p = P[a * Q + b] / (pt || 1); if (!p) continue; con += p * (a - b) ** 2; hom += p / (1 + Math.abs(a - b)); ene += p * p; ent -= p * Math.log2(p); }
    const features = { poreFrac, poreDensity, poreSizeCV, smoothFrac, texMean, wetFrac, gridCV, lacunarity, glcmContrast: con, glcmHomog: hom, glcmEnergy: ene, glcmEntropy: ent };
    // quality flags
    const flags = [];
    if (q.sharpness < 2.5) flags.push('الصورة غير حادة (مهزوزة)');
    if (q.meanL < 55) flags.push('الإضاءة ضعيفة');
    if (q.clipped > 0.08) flags.push('إضاءة زائدة / انعكاس');
    if (fracArea < 0.12) flags.push('الرغيف صغير في الصورة — قرّب الجوال');
    if (fracArea > 0.85) flags.push('الرغيف يتجاوز إطار الصورة');
    return { ok: true, features, quality: q, flags, maps: { w, h, core, pore, wet, smoothMap: tex, rgba: c.data } };
  }

  // ---------- default rule-based score (before training) ----------
  const cl = (v, a, b) => Math.max(0, Math.min(1, (v - a) / (b - a)));
  function ruleScore(f) {

/** الدمج الموزون: وزن كل مراقب من دقته على الصور المرجعية؛ رأي المصوّر ×0.8؛ حسم المشرف يتقدم على الجميع */
function computeConsensus(D) {
  const byR = {};
  D.ratings.filter(r => r.gold && r.v > 0).forEach(r => (byR[r.rater] = byR[r.rater] || []).push(r));
  const names = [...new Set(D.ratings.map(r => r.rater).concat(D.samples.map(s => s.monitor)))];
  const raters = names.map(n => {
    const g = byR[n] || [], k = g.length;
    const mae = k ? Engine.mean(g.map(x => Math.abs(x.v - x.g))) : null;
    return { name: n, n: k, acc: k ? Engine.mean(g.map(x => +(x.v === x.g))) : null, w1: k ? Engine.mean(g.map(x => +(Math.abs(x.v - x.g) <= 1))) : null, bias: k ? Engine.mean(g.map(x => x.v - x.g)) : null, w: k >= 3 ? Math.max(0.2, Math.min(1, 1 - mae / 2)) : 0.6 };
  }).sort((a, b) => (b.n ? b.w : 0) - (a.n ? a.w : 0));
  const W = Object.fromEntries(raters.map(r => [r.name, r.w]));
  const R = {}; D.ratings.filter(r => !r.gold).forEach(r => (R[r.id] = R[r.id] || []).push(r));
  const loaves = D.samples.map(s => {
    const rs = R[s.id] || [], arb = rs.filter(r => r.arb && r.v > 0).pop();
    const bad = rs.filter(r => r.v === 0).length;
    const votes = [{ v: s.own, w: (W[s.monitor] || 0.6) * 0.8 }].concat(rs.filter(r => r.v > 0 && !r.arb).map(r => ({ v: r.v, w: W[r.rater] || 0.6 }))).filter(x => x.v > 0);
    const o = { ...s, votes, n: votes.length };
    if (arb) return { ...o, label: arb.v, status: 'arb' };
    if (bad >= 2) return { ...o, status: 'invalid' };
    if (votes.length < 3) return { ...o, status: 'wait' };
    const vs = votes.map(x => x.v), spread = Math.max(...vs) - Math.min(...vs);
    if (spread >= 3) return { ...o, status: 'disputed', disputed: true };
    return { ...o, label: wMedian(votes), status: spread === 2 ? 'ok-diff' : 'ok' };
  });
  return { raters, loaves };
}
function wMedian(v) { const s = v.slice().sort((a, b) => a.v - b.v), T = s.reduce((a, b) => a + b.w, 0); let c = 0; for (const x of s) { c += x.w; if (c >= T / 2) return x.v; } return s[s.length - 1].v; }

function metricsOf(yt, yp) {
  if (!yt.length) return null;
  const idx = yt.map((t, i) => t <= 2 ? i : -1).filter(i => i >= 0);
  return { n: yt.length, accuracy: Engine.mean(yt.map((t, i) => +(t === yp[i]))), within1: Engine.mean(yt.map((t, i) => +(Math.abs(t - yp[i]) <= 1))), mae: Engine.mean(yt.map((t, i) => Math.abs(t - yp[i]))), qwk: Engine.weightedKappa(yt, yp), recallPoor: idx.length ? Engine.mean(idx.map(i => +(yp[i] <= 2))) : null };
}

  const out = $('#tr-out'); out.innerHTML = '<div class="spin">جارٍ التدريب والاختبار…</div>'; await tick();
  const ref = refRows();
  let cvNew = null, cvPub = null;
  if (lab.length >= 20) {
    const samples = [...new Set(lab.map(l => l.sampleId))], fold = Object.fromEntries(samples.map((s, i) => [s, i % 5]));
    const yt = [], yn = [];
    for (let f = 0; f < 5; f++) {
      const tr = lab.filter(l => fold[l.sampleId] !== f), te = lab.filter(l => fold[l.sampleId] === f); if (!te.length) continue;
      const m = Engine.trainModel(ref.map(r => r.f).concat(tr.map(l => l.f)), ref.map(r => r.g).concat(tr.map(l => l.label)), 1).model;
      te.forEach(l => { yt.push(l.label); yn.push(Engine.scoreWithModel(m, l.f).cls); });
      await tick();
    }
    cvNew = metricsOf(yt, yn);
    cvPub = metricsOf(lab.map(l => l.label), lab.map(l => Engine.scoreWithModel(pubModel(), l.f).cls));
  }
  const res = Engine.trainModel(ref.map(r => r.f).concat(lab.map(l => l.f)), ref.map(r => r.g).concat(lab.map(l => l.label)), 5);
  NEW = { model: { ...res.model, builtin: false }, metrics: cvNew || res.metrics, field: lab.length };
  const better = !cvNew || !cvPub || (cvNew.within1 >= cvPub.within1 - 0.02 && cvNew.qwk >= cvPub.qwk - 0.02);
  const row = (k, a, b, f = pct) => `<tr><td>${k}</td><td>${a == null ? '—' : f(a)}</td><td>${b == null ? '—' : f(b)}</td></tr>`;
  out.innerHTML = `${cvNew ? `<h3>المقارنة على الأرغفة الميدانية (${cvNew.n})</h3><table class="grid"><tr><th>المقياس</th><th>المنشور ${pubVer()}</th><th>الجديد</th></tr>
      ${row('تطابق تام', cvPub.accuracy, cvNew.accuracy)}${row('فرق ≤ 1', cvPub.within1, cvNew.within1)}${row('كشف الضعيف', cvPub.recallPoor, cvNew.recallPoor)}${row('كابا', cvPub.qwk, cvNew.qwk, v => fmt(v))}</table>`
      : '<div class="warn">أقل من 20 رغيفًا ميدانيًا معتمدًا: لا يمكن المقارنة بعد. النموذج الجديد = المرجعية + ما توفر.</div>'}
    <h3>أهم المؤشرات في النموذج الجديد</h3>${res.importance.slice(0, 6).map(i => `<div class="bar"><span>${FEATNAME[i.f] || i.f}</span><div><i class="${i.w < 0 ? 'neg' : ''}" style="width:${Math.min(100, Math.abs(i.w) * 80)}%"></i></div><b>${i.w > 0 ? '+' : ''}${fmt(i.w)}</b></div>`).join('')}
    <div class="${better ? 'hint' : 'warn'}">${better ? '✅ النموذج الجديد أدق أو مساوٍ للمنشور.' : '⚠️ النموذج الجديد أضعف من المنشور. لا يُنصح بنشره.'}</div>
    <label>ملاحظة الإصدار<input id="pub-note" placeholder="مثال: إضافة عيّنات ديسمبر"></label>
    <button class="primary" id="pub" ${better ? '' : 'disabled'}>اعتماد ونشر للمراقبين</button>${better ? '' : '<button class="danger" id="pub-force">نشر رغم ذلك</button>'}`;
  const pub = async () => {
    if (!confirm('سيصل هذا النموذج لكل المراقبين عند فتح التطبيق. متابعة؟')) return;
    try { const r = await api('publishModel', PIN, NEW.model, NEW.metrics, $('#pub-note').value, ME.name); CFG = await api('getConfig'); $('#who').textContent = ME.name + ' • النموذج ' + pubVer(); alert('تم النشر: الإصدار v' + r.version); renderAdmin(); }
    catch (e) { alert(e.message); }
  };
  $('#pub').onclick = pub; if ($('#pub-force')) $('#pub-force').onclick = pub;
}

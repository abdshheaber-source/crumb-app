<script>/* ===== Crumb analysis & statistics engine (pure JS, no DOM) ===== */
const Engine = (() => {
  // ---------- colour ----------
  function toLab(rgba, n) {
    const L = new Float32Array(n), A = new Float32Array(n), B = new Float32Array(n);
    const lin = new Float32Array(256);
    for (let i = 0; i < 256; i++) { const c = i / 255; lin[i] = c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4); }
    const f = t => t > 0.008856 ? Math.cbrt(t) : 7.787 * t + 16 / 116;
    for (let i = 0; i < n; i++) {
      const r = lin[rgba[4 * i]], g = lin[rgba[4 * i + 1]], b = lin[rgba[4 * i + 2]];
      const x = (0.4124 * r + 0.3576 * g + 0.1805 * b) / 0.95047, y = 0.2126 * r + 0.7152 * g + 0.0722 * b, z = (0.0193 * r + 0.1192 * g + 0.9505 * b) / 1.08883;
      const fx = f(x), fy = f(y), fz = f(z);
      L[i] = 116 * fy - 16; A[i] = 500 * (fx - fy); B[i] = 200 * (fy - fz);
    }
    return { L, A, B };
  }
  // ---------- filters ----------
  function boxBlur(src, w, h, r) {
    const tmp = new Float32Array(w * h), out = new Float32Array(w * h), d = 2 * r + 1;
    for (let y = 0; y < h; y++) {
      let s = 0; const o = y * w;
      for (let x = -r; x <= r; x++) s += src[o + Math.min(w - 1, Math.max(0, x))];
      for (let x = 0; x < w; x++) { tmp[o + x] = s / d; s += src[o + Math.min(w - 1, x + r + 1)] - src[o + Math.max(0, x - r)]; }
    }
    for (let x = 0; x < w; x++) {
      let s = 0;
      for (let y = -r; y <= r; y++) s += tmp[Math.min(h - 1, Math.max(0, y)) * w + x];
      for (let y = 0; y < h; y++) { out[y * w + x] = s / d; s += tmp[Math.min(h - 1, y + r + 1) * w + x] - tmp[Math.max(0, y - r) * w + x]; }
    }
    return out;
  }
  const gauss = (src, w, h, s) => { const r = Math.max(1, Math.round(s * 0.9)); return boxBlur(boxBlur(boxBlur(src, w, h, r), w, h, r), w, h, r); };
  function minmax(src, w, h, r, isMax) { // separable square min/max filter
    const tmp = new Float32Array(w * h), out = new Float32Array(w * h), op = isMax ? Math.max : Math.min;
    for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) { let v = src[y * w + x]; for (let k = Math.max(0, x - r); k <= Math.min(w - 1, x + r); k++) v = op(v, src[y * w + k]); tmp[y * w + x] = v; }
    for (let x = 0; x < w; x++) for (let y = 0; y < h; y++) { let v = tmp[y * w + x]; for (let k = Math.max(0, y - r); k <= Math.min(h - 1, y + r); k++) v = op(v, tmp[k * w + x]); out[y * w + x] = v; }
    return out;
  }
  function otsu(vals) {
    let mn = Infinity, mx = -Infinity; for (const v of vals) { if (v < mn) mn = v; if (v > mx) mx = v; }
    const bins = 256, hist = new Float64Array(bins), sc = (bins - 1) / (mx - mn || 1);
    for (const v of vals) hist[Math.round((v - mn) * sc)]++;
    let tot = vals.length, sum = 0; for (let i = 0; i < bins; i++) sum += i * hist[i];
    let wB = 0, sB = 0, best = 0, th = 0;
    for (let i = 0; i < bins; i++) { wB += hist[i]; if (!wB) continue; const wF = tot - wB; if (!wF) break; sB += i * hist[i]; const mB = sB / wB, mF = (sum - sB) / wF, v = wB * wF * (mB - mF) ** 2; if (v > best) { best = v; th = i; } }
    return mn + th / sc;
  }
  function components(mask, w, h) { // 4-connected labelling; returns {lab, sizes}
    const lab = new Int32Array(w * h), sizes = [0], st = new Int32Array(w * h);
    let n = 0;
    for (let i = 0; i < w * h; i++) {
      if (!mask[i] || lab[i]) continue;
      n++; let sp = 0, cnt = 0; st[sp++] = i; lab[i] = n;
      while (sp) { const p = st[--sp]; cnt++; const x = p % w, y = (p / w) | 0;
        if (x > 0 && mask[p - 1] && !lab[p - 1]) { lab[p - 1] = n; st[sp++] = p - 1; }
        if (x < w - 1 && mask[p + 1] && !lab[p + 1]) { lab[p + 1] = n; st[sp++] = p + 1; }
        if (y > 0 && mask[p - w] && !lab[p - w]) { lab[p - w] = n; st[sp++] = p - w; }
        if (y < h - 1 && mask[p + w] && !lab[p + w]) { lab[p + w] = n; st[sp++] = p + w; } }
      sizes.push(cnt);
    }
    return { lab, sizes };
  }
  function fillHoles(mask, w, h) {
    const inv = new Uint8Array(w * h); for (let i = 0; i < w * h; i++) inv[i] = mask[i] ? 0 : 1;
    const { lab } = components(inv, w, h), edge = new Set();
    for (let x = 0; x < w; x++) { edge.add(lab[x]); edge.add(lab[(h - 1) * w + x]); }
    for (let y = 0; y < h; y++) { edge.add(lab[y * w]); edge.add(lab[y * w + w - 1]); }
    const out = new Uint8Array(w * h); for (let i = 0; i < w * h; i++) out[i] = mask[i] || !edge.has(lab[i]) ? 1 : 0;
    return out;
  }
  const erode = (m, w, h, r) => { const f = new Float32Array(m); const o = minmax(f, w, h, r, false); const u = new Uint8Array(w * h); for (let i = 0; i < w * h; i++) u[i] = o[i] > 0.5; return u; };
  const dilate = (m, w, h, r) => { const f = new Float32Array(m); const o = minmax(f, w, h, r, true); const u = new Uint8Array(w * h); for (let i = 0; i < w * h; i++) u[i] = o[i] > 0.5; return u; };

  // ---------- segmentation: single loaf on a light board ----------
  function segment(lab, w, h, ref, roi) {
    // background model learned from the image border (works on white paper or any plain background)
    const Ls = gauss(lab.L, w, h, 2), As = gauss(lab.A, w, h, 2), Bs = gauss(lab.B, w, h, 2);
    const e = Math.max(3, Math.round(0.04 * Math.min(w, h))), bl = [], ba = [], bb = [];
    for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) if (x < e || y < e || x >= w - e || y >= h - e) { const i = y * w + x; bl.push(Ls[i]); ba.push(As[i]); bb.push(Bs[i]); }
    const med = a => { const s = a.slice().sort((p, q) => p - q); return s[s.length >> 1]; };
    const mad = (a, m) => med(a.map(v => Math.abs(v - m))) * 1.4826;
    let mL = med(bl), mA = med(ba), mB = med(bb), sL = Math.max(4, mad(bl, mL)), sA = Math.max(1.5, mad(ba, mA)), sB = Math.max(2, mad(bb, mB));
    if (ref) { mL = ref.L; mA = ref.a; mB = ref.b; sL = ref.sL || 5; sA = ref.sA || 2; sB = ref.sB || 3; }
    const d = new Float32Array(w * h), vals = [], loc = ref && ref.local;
    if (loc) sL = Math.max(4, 2 * Math.max(...loc.map(c => c.sL)));
    for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) {
      const i = y * w + x; let rL = mL, rA = mA, rB = mB;
      if (loc) { let sw = 0; rL = rA = rB = 0; for (const c of loc) { const q = 1 / ((x / w - c.x) ** 2 + (y / h - c.y) ** 2 + 1e-4); sw += q; rL += q * c.L; rA += q * c.a; rB += q * c.b; } rL /= sw; rA /= sw; rB /= sw; }
      const a = (Ls[i] - rL) / sL, b = (As[i] - rA) / sA, c = (Bs[i] - rB) / sB; d[i] = Math.sqrt(a * a + b * b + c * c); }
    for (let i = 0; i < w * h; i += 3) if (!roi || roi[i]) vals.push(Math.min(d[i], 30));
    const th = Math.max(3, otsu(vals));
    let m = new Uint8Array(w * h);
    for (let i = 0; i < w * h; i++) m[i] = d[i] > th && (!roi || roi[i]) ? 1 : 0;
    m = erode(dilate(m, w, h, 3), w, h, 3);   // close pores/gaps inside the crumb
    m = dilate(erode(m, w, h, 2), w, h, 2);   // remove specks
    const { lab: cl, sizes } = components(m, w, h);
    let best = 0; for (let i = 1; i < sizes.length; i++) if (sizes[i] > (sizes[best] || 0)) best = i;
    const out = new Uint8Array(w * h); for (let i = 0; i < w * h; i++) out[i] = cl[i] === best ? 1 : 0;
    return fillHoles(out, w, h);
  }


  // ---------- calibration sheet: 4 black corner markers (+ grey strip) on white paper ----------
  // marker centres 15 mm from each paper edge => A4: 180 x 267 mm, A3: 267 x 390 mm
  const SHEETS = { A4: [180, 267], A3: [267, 390] }, GREY_L0 = 50;
  function detectSheet(img, resample, paper) {
    const M = resample(img, null, 900), W = M.width, H = M.height, n = W * H, lb = toLab(M.data, n);
    const dk = new Uint8Array(n); for (let i = 0; i < n; i++) dk[i] = lb.L[i] < 40 && Math.abs(lb.A[i]) < 20 && Math.abs(lb.B[i]) < 20 ? 1 : 0;
    const { lab: cl, sizes } = components(dk, W, H), K = sizes.length;
    const st = Array.from({ length: K }, () => ({ n: 0, sx: 0, sy: 0, x0: 1e9, y0: 1e9, x1: -1, y1: -1 }));
    for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) { const c = cl[y * W + x]; if (!c) continue; const o = st[c]; o.n++; o.sx += x; o.sy += y; if (x < o.x0) o.x0 = x; if (x > o.x1) o.x1 = x; if (y < o.y0) o.y0 = y; if (y > o.y1) o.y1 = y; }
    const cand = [];
    for (let c = 1; c < K; c++) { const o = st[c], bw = o.x1 - o.x0 + 1, bh = o.y1 - o.y0 + 1; if (o.n < 40 || o.n > 0.01 * n) continue; const ar = bw / bh, fill = o.n / (bw * bh); if (ar < 0.6 || ar > 1.6 || fill < 0.75) continue; cand.push({ x: o.sx / o.n, y: o.sy / o.n, a: o.n, s: Math.sqrt(o.n) }); }
    const none = { found: false };
    if (cand.length < 4) return none;
    const corners = [[0, 0], [W, 0], [W, H], [0, H]], pick = [];
    for (const [cx, cy] of corners) { let best = null, bd = 1e18; for (const c of cand) { if (pick.includes(c)) continue; const d = (c.x - cx) ** 2 + (c.y - cy) ** 2; if (d < bd) { bd = d; best = c; } } pick.push(best); }
    const areas = pick.map(p => p.a).sort((a, b) => a - b); if (areas[3] > 3 * areas[0]) return none;
    const dist = (p, q) => Math.hypot(p.x - q.x, p.y - q.y);
    const top = dist(pick[0], pick[1]), bot = dist(pick[3], pick[2]), lef = dist(pick[0], pick[3]), rig = dist(pick[1], pick[2]);
    const hz = (top + bot) / 2, vt = (lef + rig) / 2, tilt = Math.max(Math.abs(top - bot) / hz, Math.abs(lef - rig) / vt);
    if (tilt > 0.25) return none;
    const sz = SHEETS[paper] || SHEETS.A4, lng = Math.max(hz, vt), sht = Math.min(hz, vt), ratio = sht / lng;
    if (ratio < 0.58 || ratio > 0.78) return none;
    const mmM = (sz[1] / lng + sz[0] / sht) / 2, ms = pick.reduce((a, p) => a + p.s, 0) / 4;
    // paper white: ring around each marker
    const corner = [], wl = [], wa = [], wb = [], gcx = pick.reduce((a, p) => a + p.x, 0) / 4, gcy = pick.reduce((a, p) => a + p.y, 0) / 4;
    for (const p of pick) { const cl0 = wl.length; for (let y = Math.max(0, Math.round(p.y - 3 * ms)); y < Math.min(H, p.y + 3 * ms); y++) for (let x = Math.max(0, Math.round(p.x - 3 * ms)); x < Math.min(W, p.x + 3 * ms); x++) {
      const r = Math.hypot(x - p.x, y - p.y); if (r < 1.3 * ms || r > 3 * ms) continue;
      if ((x - p.x) * (gcx - p.x) <= 0 || (y - p.y) * (gcy - p.y) <= 0) continue;   // inward quadrant only (stay on the paper)
      const i = y * W + x; if (lb.L[i] > 55) { wl.push(lb.L[i]); wa.push(lb.A[i]); wb.push(lb.B[i]); } }
      corner.push({ x: p.x / W, y: p.y / H, n: wl.length - cl0, L: wl.slice(cl0), a: wa.slice(cl0), b: wb.slice(cl0) }); }
    const med = a => { const s = a.slice().sort((p, q) => p - q); return s.length ? s[s.length >> 1] : NaN; };
    const madv = (a, m) => med(a.map(z => Math.abs(z - m))) * 1.4826;
    const white = { L: med(wl), a: med(wa), b: med(wb) };
    white.local = corner.map(c => ({ x: c.x, y: c.y, L: c.n ? med(c.L) : med(wl), a: c.n ? med(c.a) : med(wa), b: c.n ? med(c.b) : med(wb), sL: c.n ? madv(c.L, med(c.L)) : 0 })); white.sL = Math.max(4, 2 * madv(wl, white.L)); white.sA = Math.max(1.5, 2 * madv(wa, white.a)); white.sB = Math.max(2, 2 * madv(wb, white.b));
    // grey strip: centred between the two markers of a short side
    const shortPairs = hz < vt ? [[0, 1], [3, 2]] : [[0, 3], [1, 2]];
    let grey = null;
    for (const [i0, i1] of shortPairs) {
      const p = pick[i0], q2 = pick[i1], mx = (p.x + q2.x) / 2, my = (p.y + q2.y) / 2, ux = (q2.x - p.x) / dist(p, q2), uy = (q2.y - p.y) / dist(p, q2);
      const hl = 12 / mmM, hw = 3 / mmM, v = [];
      for (let t = -hl; t <= hl; t += 1) for (let u = -hw; u <= hw; u += 1) { const x = Math.round(mx + ux * t - uy * u), y = Math.round(my + uy * t + ux * u); if (x >= 0 && y >= 0 && x < W && y < H) v.push(lb.L[y * W + x]); }
      if (!v.length) continue; const gm = med(v), mad = med(v.map(z => Math.abs(z - gm)));
      if (gm > 25 && gm < 80 && mad < 4 && gm < white.L - 15) { grey = gm; break; }
    }
    return { found: true, paper: paper in SHEETS ? paper : 'A4', mmPerPx: mmM * (W / img.width), tilt, white, grey, markerPx: ms * img.width / W, pts: pick.map(p => [p.x / W, p.y / H]) };
  }

  // ---------- full analysis of one loaf image ----------
  // img = {data: RGBA Uint8ClampedArray, width, height}; resample: function(img, box, maxDim) -> img (provided by caller)
  function analyze(img, resample, opts = {}) {
    const sheet = detectSheet(img, resample, opts.paper || 'A4');
    // pass 1: low-res segmentation to locate the loaf
    const small = resample(img, null, 360);
    const sl = toLab(small.data, small.width * small.height);
    let roi = null;
    if (sheet.found) {   // region inside the marker quadrilateral (ignore the table around the paper)
      const P = sheet.pts.map(([u, v]) => [u * small.width, v * small.height]);
      const inside = (x, y) => { let c = false; for (let i = 0, j = 3; i < 4; j = i++) { const [xi, yi] = P[i], [xj, yj] = P[j]; if ((yi > y) !== (yj > y) && x < (xj - xi) * (y - yi) / (yj - yi) + xi) c = !c; } return c; };
      roi = new Uint8Array(small.width * small.height); for (let y = 0; y < small.height; y++) for (let x = 0; x < small.width; x++) roi[y * small.width + x] = inside(x, y) ? 1 : 0;
    }
    let sm = segment(sl, small.width, small.height, sheet.found ? sheet.white : null, roi);
    if (roi) {
      for (let i = 0; i < sm.length; i++) if (!roi[i]) sm[i] = 0;
      const { lab: c2, sizes: z2 } = components(sm, small.width, small.height); let b2 = 0; for (let i = 1; i < z2.length; i++) if (z2[i] > (z2[b2] || 0)) b2 = i;
      for (let i = 0; i < sm.length; i++) sm[i] = c2[i] === b2 && b2 ? 1 : 0;
    }
    let x0 = 1e9, y0 = 1e9, x1 = -1, y1 = -1, area = 0;
    for (let y = 0; y < small.height; y++) for (let x = 0; x < small.width; x++) if (sm[y * small.width + x]) { area++; if (x < x0) x0 = x; if (x > x1) x1 = x; if (y < y0) y0 = y; if (y > y1) y1 = y; }
    const fracArea = area / (small.width * small.height);
    const q = { loafArea: fracArea };
    if (area < 50) return { ok: false, reason: 'لم يتم العثور على رغيف في الصورة', quality: q };
    const s = img.width / small.width, pad = 6;
    const box = { x: Math.max(0, (x0 - pad) * s), y: Math.max(0, (y0 - pad) * s), w: Math.min(img.width, (x1 + pad) * s) - Math.max(0, (x0 - pad) * s), h: Math.min(img.height, (y1 + pad) * s) - Math.max(0, (y0 - pad) * s) };
    // pass 2: normalised crop (loaf ~ 600 px) => scale-invariant features
    const c = resample(img, box, 600), w = c.width, h = c.height, n = w * h;
    const lab = toLab(c.data, n);
    if (sheet.found && sheet.white.L > 50) {          // white-balance + contrast normalisation from the paper (and grey strip)
      const Lw = sheet.white.L, gain = sheet.grey != null && Lw - sheet.grey > 10 ? (100 - GREY_L0) / (Lw - sheet.grey) : 100 / Lw;
      for (let i = 0; i < n; i++) { lab.L[i] = sheet.grey != null ? 100 - (Lw - lab.L[i]) * gain : lab.L[i] * gain; lab.A[i] -= sheet.white.a; lab.B[i] -= sheet.white.b; }
    }
    const mm = sheet.found ? sheet.mmPerPx * (box.w / w) : null;   // mm per pixel of the normalised crop
    let mask = segment(lab, w, h);
    const D = Math.sqrt(mask.reduce((a, b) => a + b, 0) * 4 / Math.PI);       // equivalent diameter (px)
    const rim = Math.max(3, Math.round(D * 0.035));
    const core = erode(mask, w, h, rim);
    let N = 0; for (let i = 0; i < n; i++) N += core[i];
    if (N < 2000) return { ok: false, reason: 'الرغيف صغير جداً في الصورة — قرّب الجوال', quality: q };
    // illumination-normalised lightness
    // illumination estimated from the loaf only (normalised convolution) so a bright/white background does not leak in
    const sg0 = Math.max(8, D / 12), Lm = new Float32Array(n), Mf = new Float32Array(n);
    for (let i = 0; i < n; i++) { Mf[i] = mask[i] ? 1 : 0; Lm[i] = mask[i] ? lab.L[i] : 0; }
    const gL = gauss(Lm, w, h, sg0), gM = gauss(Mf, w, h, sg0), bg = new Float32Array(n);
    for (let i = 0; i < n; i++) bg[i] = gM[i] > 0.05 ? gL[i] / gM[i] : 50;
    const Ln = new Float32Array(n); for (let i = 0; i < n; i++) Ln[i] = lab.L[i] / (bg[i] + 1e-3);
    // quality metrics
    let sL = 0, clip = 0, lapV = 0, lapM = 0;
    for (let y = 1; y < h - 1; y++) for (let x = 1; x < w - 1; x++) { const i = y * w + x; if (!core[i]) continue; sL += lab.L[i]; if (lab.L[i] > 97) clip++; const lp = 4 * lab.L[i] - lab.L[i - 1] - lab.L[i + 1] - lab.L[i - w] - lab.L[i + w]; lapM += lp; lapV += lp * lp; }
    q.meanL = sL / N; q.clipped = clip / N; lapM /= N; q.sharpness = lapV / N - lapM * lapM;
    // pores: morphological black-hat at the pore scale
    const k = Math.max(2, Math.round(D / 85));
    const Ls = new Float32Array(n); for (let i = 0; i < n; i++) Ls[i] = Ln[i] * 120;
    const close = minmax(minmax(Ls, w, h, k, true), w, h, k, false);
    const pore = new Uint8Array(n); let pc = 0;
    for (let i = 0; i < n; i++) { const bh = close[i] - Ls[i]; pore[i] = core[i] && bh > 14 ? 1 : 0; pc += pore[i]; }
    const poreFrac = pc / N;
    const { sizes: ps } = components(pore, w, h); const pSizes = ps.slice(1).filter(v => v >= 2);
    const poreDensity = pSizes.length / N * 10000;                              // pores per 10k px
    const pm = pSizes.length ? pSizes.reduce((a, b) => a + b, 0) / pSizes.length : 0;
    const psd = pSizes.length ? Math.sqrt(pSizes.reduce((a, b) => a + (b - pm) ** 2, 0) / pSizes.length) : 0;
    const poreSizeCV = pm ? psd / pm : 0;
    // local texture (std) -> smooth fraction
    const r2 = Math.max(3, Math.round(D / 60));
    const mu = boxBlur(Ln, w, h, r2), mu2 = boxBlur(Ln.map(v => v * v), w, h, r2);
    let smooth = 0, texSum = 0; const tex = new Float32Array(n);
    for (let i = 0; i < n; i++) { if (!core[i]) continue; const sd = Math.sqrt(Math.max(0, mu2[i] - mu[i] * mu[i])); tex[i] = sd; texSum += sd; if (sd < 0.018) smooth++; }
    const smoothFrac = smooth / N, texMean = texSum / N;
    // wet / grey patches: darker AND greyer than loaf median, large blobs only
    const Lb = gauss(lab.L, w, h, Math.max(4, D / 40)), Bb = gauss(lab.B, w, h, Math.max(4, D / 40));
    const lv = [], bv = []; for (let i = 0; i < n; i += 2) if (core[i]) { lv.push(Lb[i]); bv.push(Bb[i]); }
    lv.sort((a, b) => a - b); bv.sort((a, b) => a - b);
    const Lmed = lv[lv.length >> 1], Bmed = bv[bv.length >> 1];
    const wetC = new Uint8Array(n); for (let i = 0; i < n; i++) wetC[i] = core[i] && Lb[i] < Lmed - 12 && Bb[i] < Bmed - 4 ? 1 : 0;
    const { lab: wl, sizes: ws } = components(wetC, w, h);
    const wet = new Uint8Array(n); let wc = 0;
    for (let i = 0; i < n; i++) if (wl[i] && ws[wl[i]] > 0.01 * N) { wet[i] = 1; wc++; }
    const wetFrac = wc / N;
    // spatial uniformity: CV of pore fraction over a 5x5 grid
    let bx0 = w, by0 = h, bx1 = 0, by1 = 0; for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) if (core[y * w + x]) { if (x < bx0) bx0 = x; if (x > bx1) bx1 = x; if (y < by0) by0 = y; if (y > by1) by1 = y; }
    const G = 5, cells = [];
    for (let gy = 0; gy < G; gy++) for (let gx = 0; gx < G; gx++) {
      let c1 = 0, c2 = 0; const ya = by0 + ((by1 - by0) * gy / G) | 0, yb = by0 + ((by1 - by0) * (gy + 1) / G) | 0, xa = bx0 + ((bx1 - bx0) * gx / G) | 0, xb = bx0 + ((bx1 - bx0) * (gx + 1) / G) | 0;
      for (let y = ya; y < yb; y++) for (let x = xa; x < xb; x++) { const i = y * w + x; if (core[i]) { c1++; c2 += pore[i]; } }
      if (c1 > 0.7 * (yb - ya) * (xb - xa)) cells.push(c2 / c1);
    }
    const cm = cells.reduce((a, b) => a + b, 0) / (cells.length || 1);
    const gridCV = cm < 0.005 ? 1 : cells.length > 3 ? Math.sqrt(cells.reduce((a, b) => a + (b - cm) ** 2, 0) / cells.length) / (cm + 1e-6) : 0;
    // lacunarity of pore mask (gliding box, integral image)
    const ii = new Float64Array((w + 1) * (h + 1));
    for (let y = 0; y < h; y++) { let row = 0; for (let x = 0; x < w; x++) { row += pore[y * w + x]; ii[(y + 1) * (w + 1) + x + 1] = ii[y * (w + 1) + x + 1] + row; } }
    const bxs = Math.max(8, Math.round(D / 12)); let m1 = 0, m2 = 0, nb = 0;
    for (let y = 0; y + bxs < h; y += 2) for (let x = 0; x + bxs < w; x += 2) {
      if (!core[(y + (bxs >> 1)) * w + x + (bxs >> 1)] || !core[y * w + x] || !core[(y + bxs) * w + x + bxs]) continue;
      const M = ii[(y + bxs) * (w + 1) + x + bxs] - ii[y * (w + 1) + x + bxs] - ii[(y + bxs) * (w + 1) + x] + ii[y * (w + 1) + x];
      m1 += M; m2 += M * M; nb++;
    }
    const lacunarity = nb && m1 ? Math.min(50, (m2 / nb) / ((m1 / nb) ** 2)) : 50;
    // ---------- pita-specific indicators ----------
    // pore shape: elongation & orientation coherence (second moments of each pore)
    const { lab: plab, sizes: psz } = components(pore, w, h), NP = psz.length;
    const mo = new Float64Array(NP * 5);
    for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) { const c = plab[y * w + x]; if (!c) continue; const o = c * 5; mo[o] += x; mo[o + 1] += y; mo[o + 2] += x * x; mo[o + 3] += y * y; mo[o + 4] += x * y; }
    let eW = 0, eS = 0, oc = 0, os = 0;
    for (let c = 1; c < NP; c++) { const m = psz[c]; if (m < 6) continue; const o = c * 5, mx = mo[o] / m, my = mo[o + 1] / m;
      const cxx = mo[o + 2] / m - mx * mx, cyy = mo[o + 3] / m - my * my, cxy = mo[o + 4] / m - mx * my, tr = cxx + cyy, dt = Math.sqrt(Math.max(0, (cxx - cyy) ** 2 + 4 * cxy * cxy));
      const l1 = (tr + dt) / 2, l2 = Math.max(1e-3, (tr - dt) / 2), el = Math.min(10, Math.sqrt(l1 / l2)), th = 0.5 * Math.atan2(2 * cxy, cxx - cyy);
      eS += m * el; eW += m; oc += m * Math.cos(2 * th); os += m * Math.sin(2 * th); }
    const cellElong = eW ? eS / eW : 1, orientCoh = eW ? Math.hypot(oc, os) / eW : 0;
    // stereology: mean chord length of crumb walls and of pores (pi*A/P), relative to loaf diameter
    let per = 0; for (let y = 0; y < h - 1; y++) for (let x = 0; x < w - 1; x++) { const i = y * w + x; if (!core[i]) continue; if (core[i + 1] && pore[i] !== pore[i + 1]) per++; if (core[i + w] && pore[i] !== pore[i + w]) per++; }
    const wallPx = per ? Math.PI * (N - pc) / per : D, porePx = per ? Math.PI * pc / per : 0;
    const wallRel = Math.min(30, wallPx / D * 100);
    // centre vs edge: pore fraction in the inner disk vs the outer ring
    let ccx = 0, ccy = 0; for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) if (mask[y * w + x]) { ccx += x; ccy += y; } const MA = mask.reduce((a, b) => a + b, 0); ccx /= MA; ccy /= MA;
    const R0 = D / 2; let ia = 0, ip = 0, oa = 0, op = 0;
    for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) { const i = y * w + x; if (!core[i]) continue; if (Math.hypot(x - ccx, y - ccy) < 0.5 * R0) { ia++; ip += pore[i]; } else { oa++; op += pore[i]; } }
    const pin = ia ? ip / ia : 0, pout = oa ? op / oa : 0, centerEdge = Math.abs(pin - pout) / ((pin + pout) / 2 + 0.01);
    // doughy gloss: bright AND very smooth spots; fibres: bright thin ridges (white top-hat)
    const open = minmax(minmax(Ls, w, h, k, false), w, h, k, true);
    let gl = 0, fb = 0;
    for (let i = 0; i < n; i++) { if (!core[i]) continue; if (Ln[i] > 1.1 && tex[i] < 0.018) gl++; if (Ls[i] - open[i] > 14) fb++; }
    const glossFrac = gl / N, fiberFrac = fb / N;
    // loaf shape: axis ratio from mask moments; size in cm when the sheet gave a scale
    let sxx = 0, syy = 0, sxy = 0; for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) if (mask[y * w + x]) { sxx += (x - ccx) ** 2; syy += (y - ccy) ** 2; sxy += (x - ccx) * (y - ccy); }
    sxx /= MA; syy /= MA; sxy /= MA; const tr2 = sxx + syy, dt2 = Math.sqrt((sxx - syy) ** 2 + 4 * sxy * sxy);
    const roundness = Math.sqrt(Math.max(0, (tr2 - dt2) / 2) / ((tr2 + dt2) / 2));
    const shape = { roundness, diameterCm: mm ? D * mm / 10 : null, majorCm: mm ? 4 * Math.sqrt((tr2 + dt2) / 2) * mm / 10 : null, minorCm: mm ? 4 * Math.sqrt(Math.max(0, (tr2 - dt2) / 2)) * mm / 10 : null, poreMm: mm && pc > 0.01 * N ? porePx * mm : null, wallMm: mm && pc > 0.01 * N ? wallPx * mm : null, poresPerCm2: mm ? pSizes.length / (N * mm * mm / 100) : null };
    // GLCM (16 grey levels, distance ~ pore scale, 4 directions)
    const Q = 16; let qmn = Infinity, qmx = -Infinity;
    for (let i = 0; i < n; i++) if (core[i]) { if (Ln[i] < qmn) qmn = Ln[i]; if (Ln[i] > qmx) qmx = Ln[i]; }
    const qv = new Uint8Array(n); for (let i = 0; i < n; i++) qv[i] = Math.min(Q - 1, Math.max(0, Math.floor((Ln[i] - qmn) / (qmx - qmn + 1e-6) * Q)));
    const P = new Float64Array(Q * Q); let pt = 0; const dd = Math.max(1, k);
    for (const [dx, dy] of [[dd, 0], [0, dd], [dd, dd], [dd, -dd]])
      for (let y = Math.max(0, -dy); y < h - Math.max(0, dy); y++) for (let x = 0; x < w - dx; x++) { const i = y * w + x, j = (y + dy) * w + x + dx; if (core[i] && core[j]) { P[qv[i] * Q + qv[j]]++; P[qv[j] * Q + qv[i]]++; pt += 2; } }
    let con = 0, hom = 0, ene = 0, ent = 0;
    for (let a = 0; a < Q; a++) for (let b = 0; b < Q; b++) { const p = P[a * Q + b] / (pt || 1); if (!p) continue; con += p * (a - b) ** 2; hom += p / (1 + Math.abs(a - b)); ene += p * p; ent -= p * Math.log2(p); }
    const features = { poreFrac, poreDensity, poreSizeCV, smoothFrac, texMean, wetFrac, gridCV, lacunarity, glcmContrast: con, glcmHomog: hom, glcmEnergy: ene, glcmEntropy: ent, cellElong, orientCoh, wallRel, centerEdge, glossFrac, fiberFrac };
    q.sheet = sheet.found ? { paper: sheet.paper, mmPerPx: sheet.mmPerPx, tilt: sheet.tilt, white: sheet.white.L, grey: sheet.grey } : null;
    // quality flags
    const flags = [];
    if (q.sharpness < 2.5) flags.push('الصورة غير حادة (مهزوزة)');
    if (q.meanL < 55) flags.push('الإضاءة ضعيفة');
    if (q.clipped > 0.08) flags.push('إضاءة زائدة / انعكاس');
    if (fracArea < 0.12) flags.push('الرغيف صغير في الصورة — قرّب الجوال');
    if (fracArea > 0.85) flags.push('الرغيف يتجاوز إطار الصورة');
    // native resolution of the loaf (pores need detail that resampling cannot recover)
    q.nativeDiam = Math.sqrt(area * 4 / Math.PI) * s;
    if (q.nativeDiam < 350) flags.push('دقة الرغيف في الصورة منخفضة — قرّب الجوال');
    // lighting evenness: background ring around the loaf (white paper should be uniform) + side-light gradient across the crumb
    const SW = small.width, SH = small.height, dil = dilate(sm, SW, SH, 10), near = dilate(sm, SW, SH, 3);
    let cx = 0, cy = 0; for (let y = 0; y < SH; y++) for (let x = 0; x < SW; x++) if (sm[y * SW + x]) { cx += x; cy += y; } cx /= area; cy /= area;
    const sec = Array.from({ length: 8 }, () => [0, 0]), bgL = [], half = [[0, 0], [0, 0], [0, 0], [0, 0]];
    for (let y = 0; y < SH; y++) for (let x = 0; x < SW; x++) {
      const i = y * SW + x;
      if (dil[i] && !near[i]) { const k = ((Math.round((Math.atan2(y - cy, x - cx) + Math.PI) / (2 * Math.PI) * 8)) % 8); sec[k][0] += sl.L[i]; sec[k][1]++; bgL.push(sl.L[i]); }
      if (sm[i]) { half[x < cx ? 0 : 1][0] += sl.L[i]; half[x < cx ? 0 : 1][1]++; half[y < cy ? 2 : 3][0] += sl.L[i]; half[y < cy ? 2 : 3][1]++; }
    }
    const sm8 = sec.filter(v => v[1] > 20).map(v => v[0] / v[1]);
    q.bgL = bgL.length ? bgL.reduce((a, b) => a + b, 0) / bgL.length : 0;
    q.ringRange = sm8.length >= 4 ? Math.max(...sm8) - Math.min(...sm8) : 0;
    const hm = half.map(v => v[1] ? v[0] / v[1] : 0);
    q.sideGrad = Math.max(Math.abs(hm[0] - hm[1]), Math.abs(hm[2] - hm[3]));
    q.whitePaper = q.bgL > 75;
    if (q.whitePaper && q.ringRange > 12) flags.push('ظل أو إضاءة غير متساوية حول الرغيف');
    else if (q.sideGrad > 9) flags.push('إضاءة جانبية — اجعل الضوء من الأعلى');
    if (!sheet.found) flags.push(q.whitePaper ? 'لم تظهر مربعات ورقة المعايرة كاملة — لا قياس بالملم' : 'ضع الرغيف على ورقة المعايرة البيضاء');
    else { if (sheet.tilt > 0.06) flags.push('الجوال غير موازٍ للورقة'); if (sheet.grey == null) flags.push('المستطيل الرمادي غير ظاهر — تصحيح الإضاءة جزئي'); }
    return { ok: true, features, quality: q, flags, shape, maps: { w, h, core, pore, wet, smoothMap: tex, rgba: c.data, plab, psz, mm } };
  }

  // ---------- default rule-based score (before training) ----------
  const cl = (v, a, b) => Math.max(0, Math.min(1, (v - a) / (b - a)));
  function ruleScore(f) {
    let s = 1 + 2 * cl(f.poreFrac, 0.04, 0.18) + 1 * cl(0.45 - f.smoothFrac, 0, 0.35) + 1 * cl(0.6 - f.gridCV, 0, 0.4) - 2 * cl(f.wetFrac, 0.05, 0.2);
    return Math.max(1, Math.min(5, s));
  }

  // ---------- statistics ----------
  const mean = a => a.reduce((x, y) => x + y, 0) / a.length;
  const sd = a => { const m = mean(a); return Math.sqrt(a.reduce((x, y) => x + (y - m) ** 2, 0) / Math.max(1, a.length - 1)); };
  const median = a => { const b = [...a].sort((x, y) => x - y), m = b.length >> 1; return b.length % 2 ? b[m] : (b[m - 1] + b[m]) / 2; };
  function bootstrapCI(a, B = 2000, seed = 7) {
    let s = seed; const rnd = () => (s = (s * 1664525 + 1013904223) % 4294967296) / 4294967296;
    const ms = []; for (let b = 0; b < B; b++) { let t = 0; for (let i = 0; i < a.length; i++) t += a[(rnd() * a.length) | 0]; ms.push(t / a.length); }
    ms.sort((x, y) => x - y); return [ms[Math.floor(0.025 * B)], ms[Math.floor(0.975 * B)]];
  }
  function weightedKappa(x, y, K = 5) { // quadratic weighted kappa, ratings 1..K
    const O = Array.from({ length: K }, () => new Float64Array(K)), hx = new Float64Array(K), hy = new Float64Array(K);
    for (let i = 0; i < x.length; i++) { O[x[i] - 1][y[i] - 1]++; hx[x[i] - 1]++; hy[y[i] - 1]++; }
    const n = x.length; let num = 0, den = 0;
    for (let i = 0; i < K; i++) for (let j = 0; j < K; j++) { const wgt = (i - j) ** 2 / (K - 1) ** 2; num += wgt * O[i][j]; den += wgt * hx[i] * hy[j] / n; }
    return den ? 1 - num / den : 1;
  }
  function icc21(M) { // M: rows=items, cols=raters (complete) -> ICC(2,1)
    const n = M.length, k = M[0].length, g = mean(M.flat());
    const rm = M.map(r => mean(r)), cm = Array.from({ length: k }, (_, j) => mean(M.map(r => r[j])));
    let ssr = 0, ssc = 0, sst = 0; rm.forEach(v => ssr += k * (v - g) ** 2); cm.forEach(v => ssc += n * (v - g) ** 2); M.flat().forEach(v => sst += (v - g) ** 2);
    const sse = sst - ssr - ssc, msr = ssr / (n - 1), msc = ssc / (k - 1), mse = sse / ((n - 1) * (k - 1));
    return (msr - mse) / (msr + (k - 1) * mse + k * (msc - mse) / n);
  }

  // ---------- ordinal logistic regression (cumulative logit), Adam + L2 ----------
  const sig = z => 1 / (1 + Math.exp(-z));
  function fitOrdinal(X, y, K = 5, opt = {}) {
    const d = X[0].length, lam = opt.l2 ?? 0.05, it = opt.iters ?? 600, lr = opt.lr ?? 0.05;
    // balanced sample weights: each grade contributes equally (avoids bias to the majority grade)
    const cnt = {}; y.forEach(c => cnt[c] = (cnt[c] || 0) + 1); const nc = Object.keys(cnt).length;
    const sw = opt.balanced === false ? null : y.map(c => y.length / (nc * cnt[c]));
    let w = new Float64Array(d), t0 = -2, dl = new Float64Array(K - 2).fill(Math.log(1.2));
    const pars = () => { const th = [t0]; for (let j = 0; j < K - 2; j++) th.push(th[j] + Math.exp(dl[j])); return th; };
    const P = d + 1 + (K - 2), m = new Float64Array(P), v = new Float64Array(P);
    for (let t = 1; t <= it; t++) {
      const th = pars(), gw = new Float64Array(d), gth = new Float64Array(K - 1);
      for (let i = 0; i < X.length; i++) {
        let eta = 0; for (let j = 0; j < d; j++) eta += w[j] * X[i][j];
        const c = y[i] - 1; // class 0..K-1
        const Fu = c < K - 1 ? sig(th[c] - eta) : 1, Fl = c > 0 ? sig(th[c - 1] - eta) : 0;
        const p = Math.max(1e-9, Fu - Fl), fu = c < K - 1 ? Fu * (1 - Fu) : 0, fl = c > 0 ? Fl * (1 - Fl) : 0;
        const wi = sw ? sw[i] : 1, dEta = wi * (fu - fl) / p; // d(-logp)/d eta (class-balanced)
        for (let j = 0; j < d; j++) gw[j] += dEta * X[i][j];
        if (c < K - 1) gth[c] -= wi * fu / p; if (c > 0) gth[c - 1] += wi * fl / p;
      }
      for (let j = 0; j < d; j++) gw[j] = gw[j] / X.length + lam * w[j];
      // chain rule to t0, dl
      const g = new Float64Array(P); for (let j = 0; j < d; j++) g[j] = gw[j];
      let acc = 0; for (let j = K - 2; j >= 0; j--) { acc += gth[j] / X.length; if (j === 0) g[d] = acc; else g[d + j] = acc * Math.exp(dl[j - 1]); }

        if (j < d) w[j] -= st; else if (j === d) t0 -= st; else dl[j - d - 1] -= st; }
    }
    return { w: Array.from(w), th: pars(), K };
  }
  function predictOrdinal(mdl, x) {
    let eta = 0; for (let j = 0; j < x.length; j++) eta += mdl.w[j] * x[j];
    const F = mdl.th.map(t => sig(t - eta)), p = []; let prev = 0;
    for (let k = 0; k < mdl.K; k++) { const cur = k < mdl.K - 1 ? F[k] : 1; p.push(cur - prev); prev = cur; }
    const ev = p.reduce((a, pk, k) => a + pk * (k + 1), 0);
    let mode = 0; p.forEach((v, k) => { if (v > p[mode]) mode = k; });
    return { p, expected: ev, cls: mode + 1 };
  }

    const mu = FEATS.map(f => mean(rows.map(r => r[f]))), sg = FEATS.map((f, j) => sd(rows.map(r => r[f])) || 1);
    return { mu, sg, X: rows.map(r => FEATS.map((f, j) => (r[f] - mu[j]) / sg[j])) };
  }
  function trainModel(rows, labels, folds = 5) {

    // stratified fold assignment
    const fold = new Array(n), byCls = {}; labels.forEach((c, i) => (byCls[c] = byCls[c] || []).push(i));

    for (let f = 0; f < folds; f++) {

      if (!te.length || tr.length < 10) continue;

      te.forEach(i => pred[i] = predictOrdinal(m, X[i]).cls);
    }
    const ok = pred.map((p, i) => p != null ? i : -1).filter(i => i >= 0);
    const yt = ok.map(i => labels[i]), yp = ok.map(i => pred[i]);
    const conf = Array.from({ length: 5 }, () => new Array(5).fill(0)); yt.forEach((t, i) => conf[t - 1][yp[i] - 1]++);
    const metrics = ok.length ? { n: ok.length, accuracy: mean(yt.map((t, i) => +(t === yp[i]))), within1: mean(yt.map((t, i) => +(Math.abs(t - yp[i]) <= 1))), mae: mean(yt.map((t, i) => Math.abs(t - yp[i]))), qwk: weightedKappa(yt, yp), confusion: conf,
      recallPoor: (() => { const idx = yt.map((t, i) => t <= 2 ? i : -1).filter(i => i >= 0); return idx.length ? mean(idx.map(i => +(yp[i] <= 2))) : null; })() } : null;
    const final = fitOrdinal(X, labels);
    const importance = FEATS.map((f, j) => ({ f, w: final.w[j] })).sort((a, b) => Math.abs(b.w) - Math.abs(a.w));
    return { model: { ...final, mu, sg, feats: FEATS, trainedAt: Date.now(), n }, metrics, importance };
  }
  function scoreWithModel(model, f) {
    const x = model.feats.map((k, j) => (f[k] - model.mu[j]) / model.sg[j]);
    return predictOrdinal(model, x);
  }
  return { analyze, detectSheet, ruleScore, trainModel, scoreWithModel, bootstrapCI, weightedKappa, icc21, mean, sd, median, FEATS };

/** الدمج الموزون: وزن كل مراقب من دقته على الصور المرجعية؛ رأي المصوّر ×0.8؛ حسم المشرف يتقدم على الجميع */
function computeConsensus(D) {
  const byR = {};
  D.ratings.filter(r => r.gold && r.v > 0).forEach(r => (byR[r.rater] = byR[r.rater] || []).push(r));
  const names = [...new Set(D.ratings.map(r => r.rater).concat(D.samples.map(s => s.monitor)))];
  const raters = names.map(n => {
    const g = byR[n] || [], k = g.length;
    const mae = k ? Engine.mean(g.map(x => Math.abs(x.v - x.g))) : null;
    return { name: n, n: k, acc: k ? Engine.mean(g.map(x => +(x.v === x.g))) : null, w1: k ? Engine.mean(g.map(x => +(Math.abs(x.v - x.g) <= 1))) : null, bias: k ? Engine.mean(g.map(x => x.v - x.g)) : null, w: k >= 3 ? Math.max(0.2, Math.min(1, 1 - mae / 2)) : 0.6 };
  }).sort((a, b) => (b.n ? b.w : 0) - (a.n ? a.w : 0));
  const W = Object.fromEntries(raters.map(r => [r.name, r.w]));
  const R = {}; D.ratings.filter(r => !r.gold).forEach(r => (R[r.id] = R[r.id] || []).push(r));
  const loaves = D.samples.map(s => {
    const rs = R[s.id] || [], arb = rs.filter(r => r.arb && r.v > 0).pop();
    const bad = rs.filter(r => r.v === 0).length;
    const votes = [{ v: s.own, w: (W[s.monitor] || 0.6) * 0.8 }].concat(rs.filter(r => r.v > 0 && !r.arb).map(r => ({ v: r.v, w: W[r.rater] || 0.6 }))).filter(x => x.v > 0);
    const o = { ...s, votes, n: votes.length };
    if (arb) return { ...o, label: arb.v, status: 'arb' };
    if (bad >= 2) return { ...o, status: 'invalid' };
    if (votes.length < 3) return { ...o, status: 'wait' };
    const vs = votes.map(x => x.v), spread = Math.max(...vs) - Math.min(...vs);
    if (spread >= 3) return { ...o, status: 'disputed', disputed: true };
    return { ...o, label: wMedian(votes), status: spread === 2 ? 'ok-diff' : 'ok' };
  });
  return { raters, loaves };
}
function wMedian(v) { const s = v.slice().sort((a, b) => a.v - b.v), T = s.reduce((a, b) => a + b.w, 0); let c = 0; for (const x of s) { c += x.w; if (c >= T / 2) return x.v; } return s[s.length - 1].v; }

function metricsOf(yt, yp) {
  if (!yt.length) return null;
  const idx = yt.map((t, i) => t <= 2 ? i : -1).filter(i => i >= 0);
  return { n: yt.length, accuracy: Engine.mean(yt.map((t, i) => +(t === yp[i]))), within1: Engine.mean(yt.map((t, i) => +(Math.abs(t - yp[i]) <= 1))), mae: Engine.mean(yt.map((t, i) => Math.abs(t - yp[i]))), qwk: Engine.weightedKappa(yt, yp), recallPoor: idx.length ? Engine.mean(idx.map(i => +(yp[i] <= 2))) : null };
}
